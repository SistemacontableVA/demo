function analizarModuloAtencion() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Recrear la hoja de diagnóstico
  let hojaReporte = ss.getSheetByName("DIAGNOSTICO_ATENCION");
  if (hojaReporte) {
    ss.deleteSheet(hojaReporte);
  }
  hojaReporte = ss.insertSheet("DIAGNOSTICO_ATENCION", 0);
  hojaReporte.clear();
  
  hojaReporte.getRange("A1").setValue("DIAGNOSTICO INTEGRAL - ESTRUCTURA HOJA DE ATENCION")
    .setFontSize(14).setFontWeight("bold").setFontColor("#083F4A");
    
  // --- SECCIÓN 1: PESTAÑAS DEL LIBRO ---
  hojaReporte.getRange("A3").setValue("1. PESTAÑAS DETECTADAS EN EL LIBRO").setFontWeight("bold");
  hojaReporte.getRange("A4:C4").setValues([["Nombre de la Pestaña", "Estado", "Total Filas Usadas"]])
    .setFontWeight("bold").setBackground("#0B5969").setFontColor("#FFFFFF");
    
  const todasLasHojas = ss.getSheets();
  let filaReporte = 5;
  
  todasLasHojas.forEach(hoja => {
    if (hoja.getName() !== "DIAGNOSTICO_ATENCION") {
      const estado = hoja.isSheetHidden() ? "Oculta" : "Visible";
      hojaReporte.getRange(filaReporte, 1, 1, 3).setValues([[hoja.getName(), estado, hoja.getLastRow()]]);
      filaReporte++;
    }
  });
  
  // --- SECCIÓN 2: ESTRUCTURA DE LA TABLA DE PACIENTES (Pestaña ATENCION) ---
  filaReporte += 2;
  hojaReporte.getRange(filaReporte, 1).setValue("2. ESTRUCTURA Y ENCABEZADOS DE PACIENTES (Fila 12 de 'ATENCION')").setFontWeight("bold");
  
  const hojaAtencion = ss.getSheetByName("ATENCION");
  if (!hojaAtencion) {
    filaReporte++;
    hojaReporte.getRange(filaReporte, 1).setValue("ERROR: No se encontró la pestaña 'ATENCION'")
      .setFontColor("red").setFontWeight("bold");
    return;
  }
  
  const ultimaColumna = hojaAtencion.getLastColumn();
  const ultimaFila = hojaAtencion.getLastRow();
  
  filaReporte++;
  hojaReporte.getRange(filaReporte, 1).setValue(`Total Registros de Pacientes: ${ultimaFila - 12} filas | Total Columnas: ${ultimaColumna}`)
    .setFontStyle("italic");
    
  filaReporte++;
  const cabeceraReporte = [["Columna #", "Letra", "Encabezado (Fila 12)", "Muestra 1 (Fila 13)", "Muestra 2 (Fila 14)", "Muestra 3 (Fila 15)"]];
  hojaReporte.getRange(filaReporte, 1, 1, 6).setValues(cabeceraReporte)
    .setFontWeight("bold").setBackground("#00A878").setFontColor("#FFFFFF");
    
  // Leer los encabezados en la fila 12 y muestras desde la fila 13
  const rangoDatos = hojaAtencion.getRange(12, 1, Math.min(4, ultimaFila - 11), ultimaColumna);
  const valores = rangoDatos.getValues();
  
  for (let col = 0; col < ultimaColumna; col++) {
    filaReporte++;
    
    const numColumna = col + 1;
    const letraColumna = columnaALetra(numColumna);
    const nombreEncabezado = valores[0][col] || `[Sin Nombre - Col ${letraColumna}]`;
    
    const muestra1 = valores[1] ? formatearValor(valores[1][col]) : "";
    const muestra2 = valores[2] ? formatearValor(valores[2][col]) : "";
    const muestra3 = valores[3] ? formatearValor(valores[3][col]) : "";
    
    hojaReporte.getRange(filaReporte, 1, 1, 6).setValues([
      [numColumna, letraColumna, nombreEncabezado, muestra1, muestra2, muestra3]
    ]);
  }
  
  hojaReporte.autoResizeColumns(1, 6);
  SpreadsheetApp.getUi().alert("¡Diagnóstico completado con éxito!\n\nRevisa la pestaña 'DIAGNOSTICO_ATENCION'.");
}

function columnaALetra(columna) {
  let temp, letra = '';
  while (columna > 0) {
    temp = (columna - 1) % 26;
    letra = String.fromCharCode(65 + temp) + letra;
    columna = (columna - temp - 1) / 26;
  }
  return letra;
}

function formatearValor(valor) {
  if (Object.prototype.toString.call(valor) === '[object Date]') {
    return Utilities.formatDate(valor, Session.getScriptTimeZone(), "dd/MM/yyyy");
  }
  return valor.toString();
}
