// ── Menú personalizado dentro de Google Sheets ─────────────

function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🟢 Gestión WhatsApp')
    .addItem('1. Generar Recordatorios de Consulta', 'generarRecordatorioConsulta')
    .addItem('2. Generar Recordatorios de Entrega',  'generarRecordatorioEntrega')
    .addSeparator()
    .addItem('🚀 Generar Ambos Reportes', 'generarAmbosReportes')
    .addToUi();
}


// ── Web App: punto de entrada desde el panel web ────────────

/**
 * Endpoint principal de la Web App.
 *
 * Parámetros GET aceptados:
 *   spreadsheetId  (obligatorio) — ID de la hoja de Google Sheets
 *   callback       (opcional)   — nombre de función para respuesta JSONP
 *
 * Ejemplos de URL:
 *   .../exec?spreadsheetId=ABC123
 *   .../exec?spreadsheetId=ABC123&callback=__recCallback_123
 *
 * Sin callback  → devuelve JSON puro
 * Con callback  → devuelve JSONP: __recCallback_123({...})
 *                 (esto evita el error CORS en el navegador)
 */
function doGet(e) {
  var params        = e.parameter || {};
  var spreadsheetId = params.spreadsheetId || '';
  var callback      = params.callback      || '';

  var resultado;

  try {
    if (!spreadsheetId) {
      resultado = { error: 'Falta el parámetro spreadsheetId.' };
    } else {
      // Abre la hoja por ID (no la activa) para soportar múltiples municipios
      var ss           = SpreadsheetApp.openById(spreadsheetId);
      var hojaAtencion = ss.getSheetByName('ATENCION');

      if (!hojaAtencion) {
        resultado = { error: "No se encontró la pestaña 'ATENCION' en la hoja indicada." };
      } else {
        var metaData = {
          empresa:       hojaAtencion.getRange('P1').getDisplayValue(),
          municipio:     hojaAtencion.getRange('F1').getDisplayValue(),
          lugar:         hojaAtencion.getRange('G4').getDisplayValue(),
          direccion:     hojaAtencion.getRange('G7').getDisplayValue(),
          referencia:    hojaAtencion.getRange('G9').getDisplayValue(),
          fechaAtencion: hojaAtencion.getRange('K6').getDisplayValue(),
          fechaEntrega:  hojaAtencion.getRange('K7').getDisplayValue()
        };

        var pacientes = obtenerPacientesBase(hojaAtencion);

        resultado = {
          meta:      metaData,
          pacientes: pacientes
        };
      }
    }
  } catch (err) {
    resultado = { error: 'Error en el servidor: ' + err.message };
  }

  var json = JSON.stringify(resultado);

  // JSONP: envolver la respuesta en la función callback
  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + json + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  // JSON puro (para pruebas directas en el navegador)
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}


// ── Generación de hojas dentro de Google Sheets ─────────────

function generarAmbosReportes() {
  generarRecordatorioConsulta();
  generarRecordatorioEntrega();
  SpreadsheetApp.getUi().alert("✅ Se generaron con éxito 'Recordatorio Consulta WS' y 'Recordatorio entrega WS'.");
}

function generarRecordatorioConsulta() {
  var ss           = SpreadsheetApp.getActiveSpreadsheet();
  var hojaAtencion = ss.getSheetByName('ATENCION');

  if (!hojaAtencion) {
    SpreadsheetApp.getUi().alert("❌ No se encontró la pestaña 'ATENCION'.");
    return;
  }

  var empresa       = hojaAtencion.getRange('P1').getDisplayValue();
  var municipio     = hojaAtencion.getRange('F1').getDisplayValue();
  var lugar         = hojaAtencion.getRange('G4').getDisplayValue();
  var direccion     = hojaAtencion.getRange('G7').getDisplayValue();
  var referencia    = hojaAtencion.getRange('G9').getDisplayValue();
  var fechaAtencion = hojaAtencion.getRange('K6').getDisplayValue();

  var pacientes = obtenerPacientesBase(hojaAtencion);
  if (!pacientes || pacientes.length === 0) return;

  var nombreHoja = 'Recordatorio Consulta WS';
  var hoja = ss.getSheetByName(nombreHoja);
  if (hoja) hoja.clear(); else hoja = ss.insertSheet(nombreHoja);

  var encabezados = [['No. Volante', 'Identificación', 'Paciente', 'Teléfono Original', 'Teléfono WhatsApp', 'Saldo', 'Estado Asistencia', 'Tipo de Mensaje', 'Mensaje Generado', 'Link WhatsApp (Haz clic)', 'Estado Envío']];
  hoja.getRange(1, 1, 1, encabezados[0].length).setValues(encabezados);
  hoja.getRange('1:1').setFontWeight('bold').setBackground('#d9ead3');

  var filas = [];
  for (var i = 0; i < pacientes.length; i++) {
    var p   = pacientes[i];
    var msg = '';
    var lnk = '';

    if (!p.telfLimpio) {
      msg = 'Sin Teléfono';
    } else {
      msg = 'Hola ' + p.paciente + ', le saludamos de ' + empresa +
            '. Le recordamos su cita de atención visual en el municipio ' + municipio +
            ', en ' + lugar + ' (' + direccion + ' - Ref: ' + referencia + ')' +
            ' el día ' + fechaAtencion + ' en horario continuo desde las 7:30 am hasta las 5:00pm. ¡Le esperamos!';
      lnk = 'https://wa.me/' + p.telfLimpio + '?text=' + encodeURIComponent(msg);
    }

    filas.push([p.volante, p.id, p.paciente, p.telfRaw, p.telfLimpio, p.saldo, p.estadoSondeo, 'Recordatorio Previo', msg, lnk, 'Pendiente']);
  }

  if (filas.length > 0) {
    hoja.getRange(2, 1, filas.length, filas[0].length).setValues(filas);

    var reglaDesplegable = SpreadsheetApp.newDataValidation()
      .requireValueInList(['Recordatorio Previo', 'Recordatorio Hoy'], true)
      .build();
    hoja.getRange(2, 8, filas.length, 1).setDataValidation(reglaDesplegable);

    hoja.autoResizeColumns(1, 11);
  }
}

function generarRecordatorioEntrega() {
  var ss           = SpreadsheetApp.getActiveSpreadsheet();
  var hojaAtencion = ss.getSheetByName('ATENCION');

  if (!hojaAtencion) {
    SpreadsheetApp.getUi().alert("❌ No se encontró la pestaña 'ATENCION'.");
    return;
  }

  var empresa      = hojaAtencion.getRange('P1').getDisplayValue();
  var lugar        = hojaAtencion.getRange('G4').getDisplayValue();
  var direccion    = hojaAtencion.getRange('G7').getDisplayValue();
  var fechaEntrega = hojaAtencion.getRange('K7').getDisplayValue();

  var pacientes = obtenerPacientesBase(hojaAtencion);
  if (!pacientes || pacientes.length === 0) return;

  var nombreHoja = 'Recordatorio entrega WS';
  var hoja = ss.getSheetByName(nombreHoja);
  if (hoja) hoja.clear(); else hoja = ss.insertSheet(nombreHoja);

  var encabezados = [['No. Volante', 'Identificación', 'Paciente', 'Teléfono Original', 'Teléfono WhatsApp', 'Tipo de Lente', 'Saldo a Cancelar', 'Mensaje Generado', 'Link WhatsApp (Haz clic)', 'Estado Envío']];
  hoja.getRange(1, 1, 1, encabezados[0].length).setValues(encabezados);
  hoja.getRange('1:1').setFontWeight('bold').setBackground('#fff2cc');

  var filas = [];
  for (var i = 0; i < pacientes.length; i++) {
    var p = pacientes[i];

    // Solo pacientes con "SI" en columna Z (compraron lentes)
    if (p.comproLentes !== 'SI') continue;

    var msg = '';
    var lnk = '';

    if (!p.telfLimpio) {
      msg = 'Sin Teléfono';
    } else {
      msg = 'Hola ' + p.paciente + ', le informamos de ' + empresa +
            ' que la entrega de sus lentes (' + p.tipoLente + ')' +
            ' está programada para el día ' + fechaEntrega +
            ' en ' + lugar + ' (' + direccion + ').' +
            ' Saldo pendiente a cancelar: $' + p.saldo;
      lnk = 'https://wa.me/' + p.telfLimpio + '?text=' + encodeURIComponent(msg);
    }

    filas.push([p.volante, p.id, p.paciente, p.telfRaw, p.telfLimpio, p.tipoLente, p.saldo, msg, lnk, 'Pendiente']);
  }

  if (filas.length > 0) {
    hoja.getRange(2, 1, filas.length, filas[0].length).setValues(filas);
    hoja.autoResizeColumns(1, 10);
  } else {
    SpreadsheetApp.getUi().alert("⚠️ No se encontraron pacientes con 'SI' en la columna Z (Lentes).");
  }
}


// ── Función base compartida por todas las funciones ─────────

/**
 * Lee y normaliza todos los pacientes desde la hoja ATENCION.
 * Los datos comienzan en la fila 13 (fila 12 = encabezados).
 *
 * Columnas leídas:
 *   F  (índice 5)  → volante
 *   G  (índice 6)  → id / identificación
 *   H  (índice 7)  → nombre del paciente
 *   I  (índice 8)  → teléfono 1
 *   J  (índice 9)  → teléfono 2 (fallback)
 *   K  (índice 10) → estado de sondeo / asistencia
 *   Z  (índice 25) → compró lentes (SI / NO)
 *   AB (índice 27) → tipo de lente
 *   AJ (índice 35) → saldo
 */
function obtenerPacientesBase(hojaAtencion) {
  var ultimaFila = hojaAtencion.getLastRow();
  if (ultimaFila < 13) return [];

  var numFilas = ultimaFila - 12;
  var datos    = hojaAtencion.getRange(13, 1, numFilas, 39).getValues();
  var codigoPaisDefault = '58';
  var pacientes = [];

  for (var i = 0; i < datos.length; i++) {
    var fila    = datos[i];
    var paciente = fila[7];
    if (!paciente) continue;

    var telf1      = fila[8];
    var telf2      = fila[9];
    var telfRaw    = telf1 ? telf1.toString() : (telf2 ? telf2.toString() : '');
    var telfLimpio = telfRaw.replace(/\D/g, '');

    if (telfLimpio.startsWith('0')) {
      telfLimpio = codigoPaisDefault + telfLimpio.substring(1);
    } else if (telfLimpio.length === 10 && !telfLimpio.startsWith(codigoPaisDefault)) {
      telfLimpio = codigoPaisDefault + telfLimpio;
    }

    pacientes.push({
      volante:      fila[5],
      id:           fila[6],
      paciente:     paciente,
      telfRaw:      telfRaw,
      telfLimpio:   telfLimpio,
      estadoSondeo: fila[10],
      comproLentes: fila[25] ? fila[25].toString().trim().toUpperCase() : '',
      tipoLente:    fila[27],
      saldo:        fila[35]
    });
  }

  return pacientes;
}
