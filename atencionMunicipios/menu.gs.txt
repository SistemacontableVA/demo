/**
 * Despliega la ventana flotante con la plantilla HTML
 */
function abrirReporteContable() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const plantilla = HtmlService.createTemplateFromFile('reporte_coordinacion_atencion 2');
  plantilla.reporteSpreadsheetId = '';
  plantilla.reporteMunicipioId = '';
  plantilla.reporteDatosJson = JSON.stringify(leerDatosReporteCoordinacion_(ss));
  plantilla.reporteTitulo = 'Reporte Contable de Jornada de Atención';
  const htmlOutput = plantilla.evaluate()
    .setWidth(960)
    .setHeight(800);

  SpreadsheetApp.getUi().showModalDialog(htmlOutput, 'Reporte Contable de Jornada de Atención');
}

/**
 * Abre el reporte operativo solicitado desde Atención Municipios.
 * La plantilla obtiene los datos después de validar la sesión de plataforma.
 */
function doGet(e) {
  const parametros = e && e.parameter ? e.parameter : {};
  const spreadsheetId = String(parametros.spreadsheetId || '').trim();
  const municipioId = String(parametros.municipioId || '').trim();
  if (parametros.action !== 'generar_reporte' ||
      !/^[A-Za-z0-9_-]+$/.test(spreadsheetId) ||
      !/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(municipioId)) {
    return HtmlService.createHtmlOutput('No se pudo abrir el reporte. Regresa a Atención Municipios e inténtalo nuevamente.')
      .setTitle('Reporte no disponible');
  }

  const plantilla = HtmlService.createTemplateFromFile('reporte_coordinacion_atencion 2');
  plantilla.reporteSpreadsheetId = spreadsheetId;
  plantilla.reporteMunicipioId = municipioId;
  plantilla.reporteDatosJson = '';
  plantilla.reporteTitulo = 'Reporte Contable de Jornada de Atención';
  return plantilla.evaluate().setTitle('Reporte Contable de Jornada de Atención');
}

/**
 * Valida la sesión Supabase y confirma que la hoja pertenece al municipio
 * visible para el usuario según las políticas RLS de su empresa.
 */
function autorizarHojaReporte_(spreadsheetId, municipioId, credenciales) {
  const baseUrl = 'https://cifevururqrlajkhveno.supabase.co';
  const clavePublica = 'sb_publishable_AScKVt33ih6geDdoLJopgQ_7hqH5E6h';
  const token = String(credenciales && credenciales.token || '').trim();
  const tenantId = String(credenciales && credenciales.tenantId || '').trim();
  if (!token || token.length > 4096 ||
      !/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(tenantId)) {
    throw new Error('La sesión de plataforma no es válida. Inicia sesión nuevamente.');
  }

  const opciones = {
    method: 'get',
    headers: {
      apikey: clavePublica,
      Authorization: 'Bearer ' + token
    },
    muteHttpExceptions: true
  };
  const respuestaUsuario = UrlFetchApp.fetch(baseUrl + '/auth/v1/user', opciones);
  if (respuestaUsuario.getResponseCode() !== 200) {
    throw new Error('La sesión de plataforma venció. Inicia sesión nuevamente.');
  }

  let usuario;
  try {
    usuario = JSON.parse(respuestaUsuario.getContentText());
  } catch (error) {
    throw new Error('No se pudo validar la sesión de plataforma.');
  }
  if (!usuario || !usuario.id) {
    throw new Error('No se pudo validar la sesión de plataforma.');
  }

  const consulta = [
    'select=id,link_hoja',
    'id=eq.' + encodeURIComponent(municipioId),
    'tenant_id=eq.' + encodeURIComponent(tenantId),
    'limit=1'
  ].join('&');
  const respuestaMunicipio = UrlFetchApp.fetch(
    baseUrl + '/rest/v1/atencion_municipios?' + consulta,
    opciones
  );
  if (respuestaMunicipio.getResponseCode() !== 200) {
    throw new Error('No se pudo verificar el acceso al municipio.');
  }

  let municipios;
  try {
    municipios = JSON.parse(respuestaMunicipio.getContentText());
  } catch (error) {
    throw new Error('No se pudo verificar el acceso al municipio.');
  }
  if (!Array.isArray(municipios) || municipios.length !== 1 ||
      String(municipios[0].id) !== municipioId ||
      extraerIdHojaReporte_(municipios[0].link_hoja) !== spreadsheetId) {
    throw new Error('No tienes acceso a la hoja de este municipio.');
  }
  return SpreadsheetApp.openById(spreadsheetId);
}

function extraerIdHojaReporte_(linkHoja) {
  const coincidencia = String(linkHoja || '').match(
    /^https:\/\/docs\.google\.com\/spreadsheets\/d\/([A-Za-z0-9_-]+)(?:\/|$)/
  );
  return coincidencia ? coincidencia[1] : '';
}

/**
 * Mapeo exacto basado en la reestructuración completa de la pestaña "Reporte Coordinacion"
 */
function obtenerDatosReporteCoordinacion(spreadsheetId, municipioId, credenciales) {
  if (!spreadsheetId || !municipioId || !credenciales) {
    throw new Error('Debes abrir el reporte desde Atención Municipios con una sesión válida.');
  }
  const ss = autorizarHojaReporte_(spreadsheetId, municipioId, credenciales);
  return leerDatosReporteCoordinacion_(ss);
}
